多线程

Q：有哪几种🔐

A：自旋锁和互斥锁

自旋锁有：**atomic、OSSpinLock、dispatch_semaphore_t**

互斥锁有：**os_unfair_lock、  pthread_mutex、@ synchronized、NSLock、NSConditionLock 、NSCondition、NSRecursiveLock**

自旋锁: 在尝试获取锁时以忙等的形式循环检查锁是否可用

互斥锁：当一个线程的任务没有执行完毕的时候，那么另一个线程会进入睡眠状态等待任务执行完毕，当上一个线程的任务执行完毕，那另外一个线程会自动唤醒然后执行任务。

优缺点：自旋锁优点：不会进行线程调度，时间片轮转等耗时操作。所以能在短时间内获得锁时，自旋锁的效率高于互斥锁。

​			   自旋锁缺点：一直占用CPU，降低CPU使用效率；自旋锁不能实现递归。



1、@synchroniized 内部封装了互斥锁 mutex

2、atomic 

3、dispatch_semaphore_t

4、NSLock 在同一个线程连续获取锁会造成死锁

5、NSRecursiveLock 在不同线程连续获取锁会造成死锁

6、os_unfair_lock 高性能互斥锁

Q: 有N个苹果，有两只猴子，一只猴子一次拿两个，另一只猴子一次拿三个，依次拿，直到不能拿为止，请使用多线程方案编写代码并打印日志，最终打印还剩下几个苹果。

方案一： 信号量实现

```Swift
let monkey1Sem = DispatchSemaphore(value: 1)
let monkey2Sem = DispatchSemaphore(value: 0)
let endSem = DispatchSemaphore(value: 0)
var totalApples = 99

DispatchQueue.global().async {
    while true {
        monkey1Sem.wait(timeout: .distantFuture)
        if (totalApples >= 2) {
            totalApples = totalApples - 2
            print("猴子 1 拿了 2 个苹果, 还剩下\(totalApples)个")
            monkey2Sem.signal()
        } else {
            endSem.signal()
            break
        }
    }
}

DispatchQueue.global().async {
    while true {
        monkey2Sem.wait(timeout: .distantFuture)
        if (totalApples >= 3) {
            totalApples = totalApples - 2
            print("猴子 2 拿了 3 个苹果, 还剩下\(totalApples)个")
            monkey1Sem.signal()
        } else {
            endSem.signal()
            break
        }
    }
}


DispatchQueue.global().async {
    endSem.wait(timeout: .distantFuture)
    print("苹果拿完了，还剩下 \(totalApples)个苹果！")
}
```



方案二： AI实现

```swift
class AppleManager {

    private var appleCount: Int
    /// 两个信号量都从 0 开始，避免释放时出现 current < original 的断言问题
    /// 开局由 start() 主动 signal A，形成 A -> B -> A -> B 的节奏
    private let semaphoreA = DispatchSemaphore(value: 0)
    private let semaphoreB = DispatchSemaphore(value: 0)
    /// 一旦某只猴子在自己回合拿不了，就结束整局并唤醒对方退出
    private var isFinished = false

    init(appleCount: Int = 23) {
        self.appleCount = appleCount
    }

    func start() {
        let total = appleCount
        isFinished = false
        print("开始：共 \(total) 个苹果，猴子A 每次拿 2 个，猴子B 每次拿 3 个，严格轮流拿（A→B→A→B…）。")

        let group = DispatchGroup()

        group.enter()
        DispatchQueue.global().async {
            self.takeAppleAsA()
            group.leave()
        }

        group.enter()
        DispatchQueue.global().async {
            self.takeAppleAsB()
            group.leave()
        }

        // 开局让 A 先执行一次
        semaphoreA.signal()

        // 等待两只猴子都结束，保证 AppleManager 生命周期覆盖整个并发过程
        group.wait()
    }

    private func takeAppleAsA() {
        let count = 2
        let name = "猴子A"
        while true {
            semaphoreA.wait()
            if !handleTurn(count: count, name: name, nextSemaphore: semaphoreB) { break }
            Thread.sleep(forTimeInterval: 0.1)
        }
    }

    private func takeAppleAsB() {
        let count = 3
        let name = "猴子B"
        while true {
            semaphoreB.wait()
            if !handleTurn(count: count, name: name, nextSemaphore: semaphoreA) { break }
            Thread.sleep(forTimeInterval: 0.1)
        }
    }

    @discardableResult
    private func handleTurn(count: Int, name: String, nextSemaphore: DispatchSemaphore) -> Bool {
        // 对方已经宣布结束：当前线程仅做收尾退出
        if isFinished {
            nextSemaphore.signal()
            return false
        }

        if appleCount >= count {
            appleCount -= count
            print("\(name) 拿走 \(count) 个苹果，剩余 \(appleCount)")
            nextSemaphore.signal()
            return true
        } else {
            print("\(name) 想拿 \(count) 个，但不够了")
            isFinished = true
            print("最终剩余苹果：\(appleCount)")
            nextSemaphore.signal()
            return false
        }
    }
}
```



方案三：队列实现

```Swift
class AppleQueueManager {

    private var appleCount = 23
    private let queue = DispatchQueue(label: "apple.queue")

    private var turn = 0   // 0:A  1:B
    
    init(appleCount: Int = 23) {
        self.appleCount = appleCount
    }

    func start() {

        DispatchQueue.global().async {
            self.takeApple(count: 2, monkey: 0, name: "猴子A")
        }

        DispatchQueue.global().async {
            self.takeApple(count: 3, monkey: 1, name: "猴子B")
        }
    }

    private func takeApple(count: Int, monkey: Int, name: String) {

        while true {

            queue.sync {

                if turn != monkey {
                    return
                }

                if appleCount >= count {

                    appleCount -= count
                    print("\(name) 拿走 \(count) 个苹果，剩余 \(appleCount)")

                    turn = (turn + 1) % 2

                } else {

                    print("\(name) 想拿 \(count)，但不够")
                    print("最终剩余苹果：\(appleCount)")
                    exit(0)
                }
            }
        }
    }
}
```
